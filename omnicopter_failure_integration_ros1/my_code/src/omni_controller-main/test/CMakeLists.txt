cmake_minimum_required(VERSION 3.10)
project(allocator_checks CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(Eigen3 REQUIRED)
find_path(NLOPT_INCLUDE_DIR nlopt.h)
find_library(NLOPT_LIBRARY NAMES nlopt)
if(NOT NLOPT_INCLUDE_DIR OR NOT NLOPT_LIBRARY)
  message(FATAL_ERROR "NLopt missing: install libnlopt-dev")
endif()
include_directories(../include ${EIGEN3_INCLUDE_DIR} ${EIGEN3_INCLUDE_DIRS} ${NLOPT_INCLUDE_DIR})
add_executable(allocator_checks allocator_checks.cpp)
target_link_libraries(allocator_checks ${NLOPT_LIBRARY})
enable_testing()
add_test(NAME allocator_checks COMMAND allocator_checks)
add_executable(offline_sim offline_sim.cpp)
target_link_libraries(offline_sim ${NLOPT_LIBRARY})

add_executable(governed_checks governed_checks.cpp)
target_link_libraries(governed_checks ${NLOPT_LIBRARY})
add_test(NAME governed_checks COMMAND governed_checks)
add_executable(governed_offline governed_offline.cpp)
target_link_libraries(governed_offline ${NLOPT_LIBRARY})
